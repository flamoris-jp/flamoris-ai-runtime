cmake_minimum_required(VERSION 3.28)
project(flamoris_ai_runtime VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(FLAMORIS_WARNINGS_AS_ERRORS "Treat project warnings as errors" OFF)
option(FLAMORIS_ENABLE_OPENCL "Build native OpenCL compute (no device probe at configure)" OFF)
option(FLAMORIS_NATIVE_TESTS "Enable native CPU fixture qualification" OFF)
option(FLAMORIS_OPENCL_TESTS "Enable real OpenCL parity qualification" OFF)
option(FLAMORIS_LOCKED_TOOLCHAIN "Require the reviewed compiler patch" OFF)
set(FLAMORIS_SANITIZER "none" CACHE STRING "none, address-undefined, or thread")
set_property(CACHE FLAMORIS_SANITIZER PROPERTY STRINGS none address-undefined thread)
include(cmake/ProjectOptions.cmake)
include(cmake/Dependencies.cmake)
find_package(Threads REQUIRED)

add_library(flamoris_runtime_domain STATIC src/domain/error.cpp src/domain/ids.cpp
    src/domain/effects.cpp src/domain/limits.cpp)
flamoris_target(flamoris_runtime_domain)

# Components remain separate static libraries with dependencies explicit. CONFIGURE_DEPENDS
# lets a focused implementation slice add a translation unit without editing a shared list.
file(GLOB COMPILER_SOURCES CONFIGURE_DEPENDS src/compiler/*.cpp)
add_library(flamoris_runtime_compiler STATIC ${COMPILER_SOURCES})
flamoris_target(flamoris_runtime_compiler)
target_include_directories(flamoris_runtime_compiler SYSTEM PRIVATE third_party)
target_link_libraries(flamoris_runtime_compiler PUBLIC flamoris_runtime_domain PRIVATE OpenSSL::Crypto)

file(GLOB NATIVE_SOURCES CONFIGURE_DEPENDS src/model/*.cpp src/compute/*.cpp)
list(FILTER NATIVE_SOURCES EXCLUDE REGEX "/inference\\.cpp$")
add_library(flamoris_runtime_native STATIC ${NATIVE_SOURCES})
flamoris_target(flamoris_runtime_native)
target_link_libraries(flamoris_runtime_native PUBLIC flamoris_runtime_domain PRIVATE OpenSSL::Crypto)
if(FLAMORIS_ENABLE_OPENCL)
    find_package(OpenCL REQUIRED)
    target_link_libraries(flamoris_runtime_native PRIVATE OpenCL::OpenCL)
    target_compile_definitions(flamoris_runtime_native PUBLIC FLAMORIS_ENABLE_OPENCL=1)
endif()

add_library(flamoris_runtime_event_store STATIC src/observation/event_store.cpp)
flamoris_target(flamoris_runtime_event_store)
target_link_libraries(flamoris_runtime_event_store PUBLIC flamoris_runtime_domain)

file(GLOB KERNEL_SOURCES CONFIGURE_DEPENDS src/kernel/*.cpp)
add_library(flamoris_runtime_kernel STATIC ${KERNEL_SOURCES})
flamoris_target(flamoris_runtime_kernel)
target_link_libraries(flamoris_runtime_kernel PUBLIC flamoris_runtime_domain flamoris_runtime_compiler
    flamoris_runtime_event_store Threads::Threads)

file(GLOB OBSERVATION_SOURCES CONFIGURE_DEPENDS src/observation/*.cpp src/transports/*.cpp src/adapters/*.cpp)
list(FILTER OBSERVATION_SOURCES EXCLUDE REGEX "/event_store\\.cpp$")
add_library(flamoris_runtime_observation STATIC ${OBSERVATION_SOURCES})
flamoris_target(flamoris_runtime_observation)
target_link_libraries(flamoris_runtime_observation PUBLIC flamoris_runtime_domain flamoris_runtime_kernel)

add_library(flamoris_runtime INTERFACE)
add_library(flamoris::runtime ALIAS flamoris_runtime)
target_link_libraries(flamoris_runtime INTERFACE flamoris_runtime_observation flamoris_runtime_native)
if(EXISTS "${PROJECT_SOURCE_DIR}/src/model/inference.cpp")
    add_library(flamoris_runtime_inference STATIC src/model/inference.cpp)
    flamoris_target(flamoris_runtime_inference)
    target_link_libraries(flamoris_runtime_inference PUBLIC flamoris_runtime_kernel flamoris_runtime_native)
    target_link_libraries(flamoris_runtime INTERFACE flamoris_runtime_inference)
endif()

# Compile every public header in isolation to forbid accidental include ordering contracts.
file(GLOB PUBLIC_HEADERS CONFIGURE_DEPENDS include/flamoris/runtime/*.hpp)
set(HEADER_SOURCES)
foreach(header IN LISTS PUBLIC_HEADERS)
    get_filename_component(name "${header}" NAME_WE)
    set(source "${CMAKE_CURRENT_BINARY_DIR}/header-check/${name}.cpp")
    file(GENERATE OUTPUT "${source}" CONTENT "#include <flamoris/runtime/${name}.hpp>\n")
    list(APPEND HEADER_SOURCES "${source}")
endforeach()
add_library(flamoris_header_check OBJECT ${HEADER_SOURCES})
flamoris_target(flamoris_header_check)

include(CTest)
if(BUILD_TESTING)
    add_library(flamoris_catch2 STATIC third_party/catch2/catch_amalgamated.cpp)
    add_library(Catch2::Catch2WithMain ALIAS flamoris_catch2)
    target_include_directories(flamoris_catch2 SYSTEM PUBLIC third_party/catch2)
    target_compile_features(flamoris_catch2 PUBLIC cxx_std_20)
    target_link_libraries(flamoris_catch2 PUBLIC Threads::Threads)
    flamoris_sanitizer(flamoris_catch2)
    file(GLOB_RECURSE OFFLINE_TESTS CONFIGURE_DEPENDS tests/unit/*.cpp tests/acceptance/*.cpp tests/observation/*.cpp tests/adapters/*.cpp tests/compiler/*.cpp tests/kernel/*.cpp)
    foreach(source IN LISTS OFFLINE_TESTS)
        flamoris_test("${source}" offline)
    endforeach()
    if(FLAMORIS_NATIVE_TESTS)
        file(GLOB NATIVE_TESTS CONFIGURE_DEPENDS tests/integration/native/*.cpp)
        foreach(source IN LISTS NATIVE_TESTS)
            if(source MATCHES "opencl" AND NOT FLAMORIS_OPENCL_TESTS)
                continue()
            endif()
            flamoris_test("${source}" native)
        endforeach()
    endif()
    if(FLAMORIS_OPENCL_TESTS AND NOT (FLAMORIS_ENABLE_OPENCL AND FLAMORIS_NATIVE_TESTS))
        message(FATAL_ERROR "OpenCL tests require FLAMORIS_ENABLE_OPENCL and FLAMORIS_NATIVE_TESTS")
    endif()
endif()
